  publico estatico vacio principal()
    Menu m = nuevo Menu()
    entero op
    imprimir(menuPrincipal())
    op = leerEntero("Seleccione una opción: ")
    hacer
      segun (op) hacer
        caso 1:
          m.menuArchivoEstudiante()
          romper
        caso 2:
          m.menuArchivoDocente()
          romper
        caso 3:
          m.menuArchivoCurso()
          romper
        caso 4:
          m.menuArchivoBoletin()
          romper
        por_defecto:
          imprimir("Opcion no encontrada")
      FinSegun
    mientras (op < 5)
  FinMetodo
